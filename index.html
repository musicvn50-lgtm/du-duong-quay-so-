// ========================================== // DU DUONG - QUAY SO 1 -> 100 // KHONG CAN GOOGLE SHEET // ========================================== const TOTAL_NUMBERS = 100; // ========================================== // KHỞI TẠO / RESET 100 SỐ // CHỈ CHẠY HÀM NÀY 1 LẦN KHI BẮT ĐẦU // ========================================== function khoiTao() { const props = PropertiesService.getScriptProperties(); const numbers = []; for (let i = 1; i <= TOTAL_NUMBERS; i++) { numbers.push(i); } const state = { available: numbers, used: {}, devices: {} }; props.setProperty( "DU_DUONG_STATE", JSON.stringify(state) ); return ContentService .createTextOutput( JSON.stringify({ success: true, message: "Đã tạo 100 số" }) ) .setMimeType(ContentService.MimeType.JSON); } // ========================================== // API QUAY SỐ // ========================================== function doGet(e) { try { const clientId = e && e.parameter && e.parameter.clientId ? String(e.parameter.clientId).trim() : ""; const callback = e && e.parameter && e.parameter.callback ? String(e.parameter.callback) : ""; // -------------------------------------- // KHÔNG CÓ MÃ THIẾT BỊ // -------------------------------------- if (!clientId) { return response({ success: false, message: "Thiếu mã thiết bị" }, callback); } // -------------------------------------- // KHÓA TOÀN BỘ QUÁ TRÌNH QUAY // -------------------------------------- const lock = LockService.getScriptLock(); lock.waitLock(30000); try { const props = PropertiesService .getScriptProperties(); let raw = props.getProperty("DU_DUONG_STATE"); // ------------------------------------ // CHƯA KHỞI TẠO // ------------------------------------ if (!raw) { return response({ success: false, message: "Hệ thống chưa khởi tạo" }, callback); } const state = JSON.parse(raw); // ------------------------------------ // MÁY NÀY ĐÃ QUAY // ------------------------------------ if ( state.devices && state.devices[clientId] !== undefined ) { const oldNumber = state.devices[clientId]; return response({ success: true, so: Number(oldNumber), daCo: true, message: "Thiết bị đã quay trước đó" }, callback); } // ------------------------------------ // HẾT 100 SỐ // ------------------------------------ if ( !state.available || state.available.length === 0 ) { return response({ success: false, message: "Đã hết 100 số" }, callback); } // ------------------------------------ // CHỌN NGẪU NHIÊN // ------------------------------------ const randomIndex = Math.floor( Math.random() * state.available.length ); const number = state.available[randomIndex]; // ------------------------------------ // XÓA SỐ KHỎI KHO // ------------------------------------ state.available.splice( randomIndex, 1 ); // ------------------------------------ // GHI SỐ ĐÃ DÙNG // ------------------------------------ if (!state.used) { state.used = {}; } state.used[number] = true; // ------------------------------------ // GHI MÁY → SỐ // ------------------------------------ if (!state.devices) { state.devices = {}; } state.devices[clientId] = number; // ------------------------------------ // LƯU LẠI // ------------------------------------ props.setProperty( "DU_DUONG_STATE", JSON.stringify(state) ); // ------------------------------------ // TRẢ KẾT QUẢ // ------------------------------------ return response({ success: true, so: Number(number), daCo: false, message: "Nhận số thành công" }, callback); } finally { lock.releaseLock(); } } catch (error) { return response({ success: false, message: error.message }, ""); } } // ========================================== // JSON / JSONP // ========================================== function response(data, callback) { const json = JSON.stringify(data); // JSONP if (callback) { return ContentService .createTextOutput( callback + "(" + json + ");" ) .setMimeType( ContentService.MimeType.JAVASCRIPT ); } // JSON return ContentService .createTextOutput(json) .setMimeType( ContentService.MimeType.JSON ); }
