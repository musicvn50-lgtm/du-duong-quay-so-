const SHEET_ID = "1aKuk2cEMV9ykDzRqf0mol-jN2Qd7hQa2ox738smF-BM"; const SHEET_NAME = "Sheet1"; function getSheet() { const ss = SpreadsheetApp.openById(SHEET_ID); let sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(SHEET_NAME); } return sheet; } // ================================ // TẠO 100 MÃ // ================================ function khoiTaoBang() { const sheet = getSheet(); sheet.clear(); sheet.getRange("A1:D1").setValues([[ "Mã", "Trạng thái", "Thời gian", "Mã thiết bị" ]]); const rows = []; for (let i = 1; i <= 100; i++) { rows.push([ String(i).padStart(3, "0"), "Chưa dùng", "", "" ]); } sheet.getRange(2, 1, 100, 4).setValues(rows); sheet.getRange("A1:D1").setFontWeight("bold"); sheet.autoResizeColumns(1, 4); } // ================================ // API QUAY SỐ // ================================ function doGet(e) { try { const clientId = e && e.parameter && e.parameter.clientId ? String(e.parameter.clientId) : ""; const callback = e && e.parameter && e.parameter.callback ? String(e.parameter.callback) : ""; if (!clientId) { return traVe({ success: false, message: "Thiếu mã thiết bị" }, callback); } const lock = LockService.getScriptLock(); lock.waitLock(30000); try { const sheet = getSheet(); const lastRow = sheet.getLastRow(); if (lastRow < 2) { return traVe({ success: false, message: "Chưa tạo 100 mã" }, callback); } const data = sheet .getRange(2, 1, lastRow - 1, 4) .getValues(); // ================================ // KIỂM TRA THIẾT BỊ ĐÃ NHẬN MÃ // ================================ for (let i = 0; i < data.length; i++) { if ( data[i][3] && String(data[i][3]) === clientId && data[i][1] === "Đã dùng" ) { return traVe({ success: true, ma: String(data[i][0]).padStart(3, "0"), daCo: true, message: "Thiết bị đã nhận mã trước đó" }, callback); } } // ================================ // TÌM CÁC MÃ CHƯA DÙNG // ================================ const available = []; for (let i = 0; i < data.length; i++) { if (data[i][1] !== "Đã dùng") { available.push(i + 2); } } // ================================ // HẾT MÃ // ================================ if (available.length === 0) { return traVe({ success: false, message: "Đã hết 100 mã may mắn" }, callback); } // ================================ // CHỌN NGẪU NHIÊN // ================================ const index = Math.floor( Math.random() * available.length ); const row = available[index]; const ma = String( sheet.getRange(row, 1).getValue() ).padStart(3, "0"); // ================================ // GHI NHẬN MÃ // ================================ sheet.getRange(row, 2) .setValue("Đã dùng"); sheet.getRange(row, 3) .setValue(new Date()); sheet.getRange(row, 4) .setValue(clientId); SpreadsheetApp.flush(); // ================================ // TRẢ KẾT QUẢ // ================================ return traVe({ success: true, ma: ma, daCo: false, message: "Nhận mã thành công" }, callback); } finally { lock.releaseLock(); } } catch (error) { return traVe({ success: false, message: error.message }, ""); } } // ================================ // TRẢ JSON / JSONP // ================================ function traVe(data, callback) { const json = JSON.stringify(data); // Có callback → trả JSONP if (callback) { return ContentService .createTextOutput( callback + "(" + json + ");" ) .setMimeType( ContentService.MimeType.JAVASCRIPT ); } // Không có callback → trả JSON bình thường return ContentService .createTextOutput(json) .setMimeType( ContentService.MimeType.JSON ); }
