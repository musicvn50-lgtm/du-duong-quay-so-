const SHEET_ID = "1aKuk2cEMV9ykDzRqf0mol-jN2Qd7hQa2ox738smF-BM"; const SHEET_NAME = "Sheet1"; // ======================================== // LẤY SHEET // ======================================== function getSheet() { const ss = SpreadsheetApp.openById(SHEET_ID); let sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(SHEET_NAME); } return sheet; } // ======================================== // KHỞI TẠO 100 SỐ // ======================================== function khoiTaoBang() { const sheet = getSheet(); sheet.clear(); sheet.getRange(1, 1, 1, 4).setValues([[ "Số", "Trạng thái", "Thời gian", "Mã lượt" ]]); const data = []; for (let i = 1; i <= 100; i++) { data.push([ i, "Chưa dùng", "", "" ]); } sheet.getRange(2, 1, 100, 4).setValues(data); sheet.getRange("A1:D1") .setFontWeight("bold"); sheet.autoResizeColumns(1, 4); } // ======================================== // QUAY SỐ // ======================================== function doGet(e) { const callback = e && e.parameter && e.parameter.callback ? String(e.parameter.callback) : ""; let clientId = e && e.parameter && e.parameter.clientId ? String(e.parameter.clientId).trim() : ""; // ====================================== // NẾU KHÔNG CÓ MÃ THIẾT BỊ // ====================================== if (!clientId) { return traVe({ success: false, message: "Thiếu mã thiết bị" }, callback); } // ====================================== // KHÓA ĐỂ 100 NGƯỜI QUAY CÙNG LÚC // KHÔNG BAO GIỜ TRÙNG SỐ // ====================================== const lock = LockService.getScriptLock(); lock.waitLock(30000); try { const sheet = getSheet(); const lastRow = sheet.getLastRow(); if (lastRow < 101) { return traVe({ success: false, message: "Chưa có đủ 100 số" }, callback); } // ==================================== // KIỂM TRA THIẾT BỊ ĐÃ QUAY // ==================================== const maLuotData = sheet .getRange(2, 4, 100, 1) .getValues(); for (let i = 0; i < maLuotData.length; i++) { if ( maLuotData[i][0] && String(maLuotData[i][0]) === clientId ) { const so = sheet .getRange(i + 2, 1) .getValue(); return traVe({ success: true, so: Number(so), daCo: true, message: "Thiết bị đã quay trước đó" }, callback); } } // ==================================== // LẤY DANH SÁCH SỐ CHƯA DÙNG // ==================================== const trangThai = sheet .getRange(2, 2, 100, 1) .getValues(); const available = []; for (let i = 0; i < 100; i++) { if (trangThai[i][0] !== "Đã dùng") { available.push(i + 2); } } // ==================================== // HẾT SỐ // ==================================== if (available.length === 0) { return traVe({ success: false, message: "Đã hết 100 số" }, callback); } // ==================================== // CHỌN NGẪU NHIÊN // ==================================== const randomIndex = Math.floor( Math.random() * available.length ); const row = available[randomIndex]; const so = Number( sheet .getRange(row, 1) .getValue() ); // ==================================== // GHI NHẬN // ==================================== sheet .getRange(row, 2, 1, 3) .setValues([[ "Đã dùng", new Date(), clientId ]]); SpreadsheetApp.flush(); // ==================================== // TRẢ KẾT QUẢ // ==================================== return traVe({ success: true, so: so, daCo: false, message: "Nhận số thành công" }, callback); } catch (error) { return traVe({ success: false, message: error.message }, callback); } finally { lock.releaseLock(); } } // ======================================== // TRẢ JSON / JSONP // ======================================== function traVe(data, callback) { const json = JSON.stringify(data); if (callback) { return ContentService .createTextOutput( callback + "(" + json + ");" ) .setMimeType( ContentService.MimeType.JAVASCRIPT ); } return ContentService .createTextOutput(json) .setMimeType( ContentService.MimeType.JSON ); }
