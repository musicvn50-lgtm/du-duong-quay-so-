const SHEET_ID = "1aKuk2cEMV9ykDzRqf0mol-jN2Qd7hQa2ox738smF-BM"; const SHEET_NAME = "Sheet1"; // ===================================================== // LẤY SHEET // ===================================================== function getSheet() { const ss = SpreadsheetApp.openById(SHEET_ID); let sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(SHEET_NAME); } return sheet; } // ===================================================== // TẠO BẢNG 100 SỐ // CHỈ CHẠY 1 LẦN // ===================================================== function khoiTaoBang() { const sheet = getSheet(); sheet.clear(); sheet.getRange(1, 1, 1, 4).setValues([[ "Số", "Trạng thái", "Thời gian", "Mã lượt" ]]); const rows = []; for (let i = 1; i <= 100; i++) { rows.push([ i, "Chưa dùng", "", "" ]); } sheet.getRange(2, 1, 100, 4) .setValues(rows); sheet.getRange(1, 1, 1, 4) .setFontWeight("bold"); sheet.autoResizeColumns(1, 4); } // ===================================================== // API // ===================================================== function doGet(e) { let callback = ""; try { const clientId = e && e.parameter && e.parameter.clientId ? String(e.parameter.clientId) : ""; callback = e && e.parameter && e.parameter.callback ? String(e.parameter.callback) : ""; // Không có mã lượt if (!clientId) { return traVe({ success: false, message: "Thiếu mã lượt" }, callback); } // Khóa để 2 người không lấy cùng 1 số const lock = LockService.getScriptLock(); lock.waitLock(30000); try { const sheet = getSheet(); const lastRow = sheet.getLastRow(); if (lastRow < 2) { return traVe({ success: false, message: "Chưa tạo bảng 100 số" }, callback); } // Đọc A:D const data = sheet .getRange( 2, 1, lastRow - 1, 4 ) .getValues(); // ================================================= // KIỂM TRA MÃ LƯỢT ĐÃ QUAY // ================================================= for (let i = 0; i < data.length; i++) { const so = data[i][0]; const trangThai = String(data[i][1]); const maLuot = String(data[i][3]); if ( trangThai === "Đã dùng" && maLuot === clientId ) { return traVe({ success: true, ma: String(so) .padStart(3, "0"), daCo: true, message: "Mã lượt đã nhận số trước đó" }, callback); } } // ================================================= // TÌM SỐ CHƯA DÙNG // ================================================= const available = []; for (let i = 0; i < data.length; i++) { if ( String(data[i][1]) !== "Đã dùng" ) { available.push(i + 2); } } // ================================================= // HẾT SỐ // ================================================= if (available.length === 0) { return traVe({ success: false, message: "Đã hết 100 số may mắn" }, callback); } // ================================================= // CHỌN SỐ // ================================================= const randomIndex = Math.floor( Math.random() * available.length ); const row = available[randomIndex]; const so = Number( sheet .getRange(row, 1) .getValue() ); const ma = String(so) .padStart(3, "0"); // ================================================= // GHI VÀO SHEET // ================================================= sheet .getRange(row, 2) .setValue("Đã dùng"); sheet .getRange(row, 3) .setValue(new Date()); sheet .getRange(row, 4) .setValue(clientId); SpreadsheetApp.flush(); // ================================================= // TRẢ KẾT QUẢ // ================================================= return traVe({ success: true, ma: ma, daCo: false, message: "Nhận mã thành công" }, callback); } finally { lock.releaseLock(); } } catch (error) { return traVe({ success: false, message: error.message }, callback); } } // ===================================================== // JSON / JSONP // ===================================================== function traVe(data, callback) { const json = JSON.stringify(data); if (callback) { return ContentService .createTextOutput( callback + "(" + json + ");" ) .setMimeType( ContentService.MimeType.JAVASCRIPT ); } return ContentService .createTextOutput(json) .setMimeType( ContentService.MimeType.JSON ); }
